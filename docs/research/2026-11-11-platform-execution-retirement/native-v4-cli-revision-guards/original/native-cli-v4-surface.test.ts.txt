interface JsonEnvelope {
  command: string | null;
  exitCode: number;
  data?: Record<string, unknown>;
  error?: { code: string; message: string };
}

function json(result: Awaited<ReturnType<typeof runNativeCli>>): JsonEnvelope {
  expect(result.stdout).toBeTruthy();
  return JSON.parse(result.stdout!) as JsonEnvelope;
}

function inputTemplate(result: JsonEnvelope, name: string): Record<string, unknown> {
  const continuation = result.data?.continuation as {
    inputOptions: Array<{ name: string; template: Record<string, unknown> }>;
  };
  const option = continuation.inputOptions.find((option) => option.name === name);
  expect(option).toBeDefined();
  expect(Array.isArray(option!.template)).toBe(false);
  return structuredClone(option!.template);
}

  async function runnerStep(name: string, input: unknown): Promise<JsonEnvelope> {
    runnerInputSequence += 1;
    let payload = input;
    if (
      input &&
      typeof input === 'object' &&
      (input as { kind?: string }).kind &&
      ['verifier-execution-error', 'verifier-unavailable'].includes(
        (input as { kind: string }).kind,
      ) &&
      !('stateVersion' in input)
    ) {
      const current = json(await runNativeCli(['show', name, '--json', ...projectArgs()]));
      const state = current.data?.state as {
        state_version: number;
        loop: { iteration: number; attempt: number };
      };
      const runtimeState = await readNativePortableChange(
        await nativeProjectPaths(projectRoot, 'docs'),
        name,
      );
      const verifierRef = runtimeState.verifier_action
        ? nativeVerifierActionExecutionRef(runtimeState.verifier_action)
        : null;
      payload = {
        ...(input as Record<string, unknown>),
        stateVersion: state.state_version,
        iteration: state.loop.iteration,
        attempt: state.loop.attempt,
        verifierExecutionRef: verifierRef,
      };
    }
    if (
      input &&
      typeof input === 'object' &&
      (input as { kind?: string }).kind === 'verifier-response' &&
      !('candidateId' in input)
    ) {
      const current = json(await runNativeCli(['show', name, '--json', ...projectArgs()]));
      const state = current.data?.state as {
        builder_handoff: { candidate_id: string };
      };
      const runtimeState = await readNativePortableChange(
        await nativeProjectPaths(projectRoot, 'docs'),
        name,
      );
      const verifierRef = runtimeState.verifier_action
        ? nativeVerifierActionExecutionRef(runtimeState.verifier_action)
        : null;
      payload = {
        ...(input as Record<string, unknown>),
        candidateId: state.builder_handoff.candidate_id,
        verifierExecutionRef: verifierRef,
      };
    }
    const file = path.join(
      resolveWorkspaceRuntimeRootSync(projectRoot),
      `.native-runner-input-${runnerInputSequence}.json`,
    );
    await fs.writeFile(file, JSON.stringify(payload));
    try {
      return json(
        await runNativeCli(['next', name, '--runner-input', file, '--json', ...projectArgs()]),
      );
    } finally {
      await fs.rm(file, { force: true });
    }
  }

  it('retries an unavailable platform Verifier after its failure budget while preserving candidate and Agent reports', async () => {
    const name = 'retry-unavailable-verifier';
    await prepareBuild(name, ['First behavior works.'], 'zh-CN');
    await runnerStep(name, {
      ...builderHandoff(['A1']),
      checks: [
        {
          name: 'SIMULATED Agent completed check',
          result: 'passed',
          note: 'Agent platform report only; no Comet command',
        },
      ],
    });
    const paths = await nativeProjectPaths(projectRoot, 'docs'),
      built = await readNativePortableChange(paths, name);
    const candidate = built.builder_handoff!.candidate_id,
      reports = structuredClone(built.builder_handoff!.checks);
    await runnerStep(name, { kind: 'dispatch-verifier', checks: [] });
    for (let attempt = 1; attempt <= 3; attempt++) {
      const failed = await runnerStep(name, {
        kind: 'verifier-unavailable',
        summary: 'SIMULATED platform task ended without a usable result.',
      });
      expect(failed.exitCode, JSON.stringify(failed)).toBe(0);
      const state = await readNativePortableChange(paths, name);
      expect(state.builder_handoff!.candidate_id).toBe(candidate);
      expect(state.builder_handoff!.checks).toEqual(reports);
      if (attempt < 3) {
        expect(state.status).toBe('active');
        await runnerStep(name, { kind: 'dispatch-verifier', checks: [] });
      } else expect(state.status).toBe('blocked');
    }
    const blocked = await readNativePortableChange(paths, name);
    const retry = json(
      await runNativeCli([
        'next',
        name,
        '--retry-verifier',
        '--summary',
        'SIMULATED USER explicitly retries the isolated platform review',
        '--expected-state-version',
        String(blocked.state_version),
        '--expected-action',
        'retry-verifier',
        '--json',
        ...projectArgs(),
      ]),
    );
    expect(retry).toMatchObject({
      exitCode: 0,
      data: {
        state: { phase: 'verify', status: 'active', loop: { retry_epoch: 1 } },
        continuation: { action: 'dispatch-verifier' },
      },
    });
    const dispatched = await runnerStep(name, {
      kind: 'dispatch-verifier',
      checks: [],
    });
    expect(dispatched).toMatchObject({
      exitCode: 0,
      data: { verifierDispatch: { candidateId: candidate, attempt: 4 } },
    });
    const after = await readNativePortableChange(paths, name);
    expect(after.builder_handoff!.checks).toEqual(reports);
    expect(after.builder_handoff!.candidate_id).toBe(candidate);
  });
