import { execFileSync } from 'node:child_process';
import { promises as fs } from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { afterEach, expect, it, vi } from 'vitest';
import { createOpenSpecArchiveFixture } from 'D:/Project/Comet-runtime-review-053/test/helpers/classic-openspec-fixture.js';
import { canRepairClassicDelivery } from 'D:/Project/Comet-runtime-review-053/domains/comet-classic/classic-sdk-delivery-repair-model.js';
import { runClassicCli } from 'D:/Project/Comet-runtime-review-053/domains/comet-classic/classic-cli.js';
import {
  defineClassicAgentWorkflowApplication,
  defineClassicWorkflowApplication,
} from 'D:/Project/Comet-runtime-review-053/domains/comet-classic/classic-sdk-application.js';
import { inspectClassicSdkRun } from 'D:/Project/Comet-runtime-review-053/domains/comet-classic/classic-sdk-status.js';

const roots: string[] = [];
afterEach(async () => {
  await Promise.all(
    roots
      .splice(0)
      .map((root) => fs.rm(root, { recursive: true, force: true, maxRetries: 5, retryDelay: 100 })),
  );
});

it.each(['full', 'hotfix', 'tweak'] as const)(
  'compiles Classic %s Agent checks without changing the legacy definition',
  (profile) => {
    const legacy = defineWorkflow(defineClassicWorkflowApplication(profile).workflow);
    const current = defineWorkflow(defineClassicAgentWorkflowApplication(profile).workflow);
    expect(legacy.version).toBe('1');
    expect(current.version).toBe('2');
    expect(
      Object.keys(current.steps).filter((id) =>
        /\.(build|verify)\.check(?:\.evidence)?$/u.test(id),
      ),
    ).toEqual([]);
    expect(Object.keys(legacy.steps).some((id) => id.endsWith('.verify.check'))).toBe(true);
    expect(current.steps[`${profile}.archive.confirm`]).toMatchObject({ type: 'ask_user' });
  },
);

it('probe v2 repair-delivery remains available after a platform-report Archive', async () => {
  const root = await fs.mkdtemp(path.join(os.tmpdir(), 'classic-agent-checks-'));
  roots.push(root);
  await prepareClassicLegacyProject(root);
  execFileSync('git', ['init', '-b', 'main'], { cwd: root, stdio: 'ignore' });
  execFileSync(
    'git',
    [
      '-c',
      'user.name=Test',
      '-c',
      'user.email=test@example.invalid',
      'commit',
      '--allow-empty',
      '-m',
      'base',
    ],
    { cwd: root, stdio: 'ignore' },
  );
  const cli = async (...args: string[]) => {
    const result = await runClassicCli([args[0], '--json', ...args.slice(1)], undefined, {
      projectRoot: root,
      invocationCwd: root,
    });
    return JSON.parse(result.stdout!);
  };
  const initialized = await cli('state', 'init', 'example', 'hotfix', '--isolation', 'current');
  expect(initialized.exitCode, initialized.stderr).toBe(0);
  const dir = path.join(root, 'openspec/changes/example');
  await fs.writeFile(path.join(dir, 'proposal.md'), '# Proposal\nFix a small defect.\n');
  await fs.writeFile(path.join(dir, 'design.md'), '# Design\nKeep the existing scope.\n');
  await fs.writeFile(path.join(dir, 'tasks.md'), '- [ ] Implement the change\n');
  const preview = await cli('guard', 'example', 'open');
  const opened = await cli(
    'guard',
    'example',
    'open',
    '--apply',
    '--approval-hash',
    preview.data.approvalHash,
  );
  expect(opened.exitCode, opened.stderr).toBe(0);
  await fs.writeFile(path.join(dir, 'tasks.md'), '- [x] Implement the change\n');
  await fs.writeFile(
    path.join(root, 'checks.json'),
    JSON.stringify({ checks: [{ name: 'Actual platform command exited 7', result: 'failed' }] }),
  );
  const before = (await inspectClassicSdkRun(root, 'example')).run;
  const managed = await cli(
    'check',
    'run',
    'example',
    'build',
    '--',
    process.execPath,
    '-e',
    'process.exit(0)',
  );
  expect(managed.exitCode).not.toBe(0);
  expect((await inspectClassicSdkRun(root, 'example')).run).toEqual(before);
  const rejected = await cli('state', 'complete-build', 'example', '--checks', 'checks.json');
  expect(rejected.exitCode).not.toBe(0);
  expect((await inspectClassicSdkRun(root, 'example')).run).toEqual(before);
  await fs.writeFile(
    path.join(root, 'checks.json'),
    JSON.stringify({
      checks: [{ name: 'Generate required output', result: 'passed', note: 'Actual platform Node command created generated/check-output.bin in projectRoot.' }],
    }),
  );
  execFileSync(process.execPath, ['-e', "const fs=require('node:fs');fs.mkdirSync('generated',{recursive:true});fs.writeFileSync('generated/check-output.bin','actual output');"], { cwd: root });
  const completed = await cli('state', 'complete-build', 'example', '--checks', 'checks.json');
  expect(completed.exitCode, completed.stderr).toBe(0);
  const next = (await inspectClassicSdkRun(root, 'example')).run;
  expect(next.workflow.version).toBe('2');
  expect(next.actions.some((action) => action.stepId.endsWith('.build.check'))).toBe(false);
  expect(next.actions.find((action) => action.status === 'pending')?.stepId).toBe(
    'hotfix.verify.run',
  );
  expect(completed.data.checksMode).toBe('agent');
  const reportRef = 'docs/superpowers/reports/example.md';
  await fs.mkdir(path.dirname(path.join(root, reportRef)), { recursive: true });
  await fs.writeFile(
    path.join(root, reportRef),
    '# Verification\nAll scoped acceptance items were reviewed.\n',
  );
  const verified = await cli(
    'guard',
    'example',
    'verify',
    '--report',
    reportRef,
    '--apply',
    '--checks',
    'checks.json',
  );
  expect(verified.exitCode, verified.stderr).toBe(0);
  const final = (await inspectClassicSdkRun(root, 'example')).run;
  expect((final.state as { phase: string }).phase).toBe('archive');
  expect(final.actions.some((action) => action.stepId.endsWith('.verify.check'))).toBe(false);
  expect(final.actions.find((action) => action.status === 'pending')?.stepId).toBe(
    'hotfix.archive.prepare',
  );
  await fs.writeFile(path.join(root, '.comet/check-policy.json'), JSON.stringify({version: 2, commands: [{argv: [process.execPath, '-e', "const fs=require('node:fs');fs.mkdirSync('generated',{recursive:true});fs.writeFileSync('generated/check-output.bin','actual output');"], cwd: '.', outputs: ['generated/**']}]}));
  const openSpec = await createOpenSpecArchiveFixture(root);
  vi.stubEnv('COMET_OPENSPEC', openSpec.command);
  const proposal = await cli('state', 'propose-archive', 'example', '--summary', 'Fixture user approves local delivery.');
  expect(proposal.exitCode, proposal.stderr).toBe(0);
  const approval = await cli('state', 'decide-archive', 'example', '--proposal-hash', proposal.data.wait.proposalHash, '--choice', 'local');
  expect(approval.exitCode, approval.stderr).toBe(0);
  const archived = await cli('guard', 'example', 'archive', '--apply');
  expect(archived.exitCode, JSON.stringify(archived)).toBe(0);
  const original = (await inspectClassicSdkRun(root, 'example')).run;
  expect(original.workflow.version).toBe('2');
  expect(canRepairClassicDelivery(original)).toBe(true);
  const repairPreview = await cli('state', 'next', 'example');
  await fs.unlink(path.join(root, 'generated/check-output.bin'));
  const repair = await cli('state', 'repair-delivery', 'example', '--expected-revision', String(original.revision));
  const after = (await inspectClassicSdkRun(root, 'example')).run;
  await fs.writeFile(path.resolve('coverage/review-053/v2-repair-evidence.json'), JSON.stringify({ workflow: original.workflow, canRepair: canRepairClassicDelivery(original), next: repairPreview, repair, originalRunPreserved: JSON.stringify(original) === JSON.stringify(after), archiveExecutions: (await openSpec.invocations()).length }, null, 2));
  expect(after).toEqual(original);
  expect(repair.exitCode, JSON.stringify(repair)).toBe(0);
  vi.unstubAllEnvs();
});
